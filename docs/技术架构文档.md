# core/keyword/engine.py

class KeywordType(Enum):
    DEMAND = "demand"      # 需求类
    INQUIRY = "inquiry"    # 咨询类
    PRICE = "price"        # 价格类
    COMPETITOR = "competitor"  # 竞品类


class KeywordStatus(Enum):
    PENDING = "pending"     # 待处理
    APPROVED = "approved"   # 已添加
    EXECUTING = "executing" # 执行中
    COMPLETED = "completed" # 已完成
    DISCARDED = "discarded" # 废弃


class Keyword:
    """关键词"""
    def __init__(self, text: str, keyword_type: KeywordType):
        self.text = text
        self.type = keyword_type
        self.status = KeywordStatus.PENDING
        self.trigger_count = 0
        self.match_pattern: str = None

    def to_regex(self) -> str:
        """转换为正则表达式"""
        return re.escape(self.text)


class KeywordEngine:
    """关键词引擎"""
    def __init__(self, db: AsyncSession):
        self.db = db
        self.keywords: Dict[str, Keyword] = {}
        self.regex_patterns: List[Pattern] = []

    async def load_keywords(self):
        """加载关键词到内存"""
        keywords = await self.db.query(Keyword).filter(
            Keyword.status.in_([KeywordStatus.APPROVED, KeywordStatus.EXECUTING])
        ).all()

        for kw in keywords:
            self.keywords[kw.text] = kw
            self.regex_patterns.append(re.compile(kw.to_regex(), re.IGNORECASE))

    async def match(self, text: str) -> List[Keyword]:
        """匹配关键词"""
        matched = []
        for pattern in self.regex_patterns:
            if pattern.search(text):
                keyword = self.keywords.get(pattern.pattern)
                if keyword:
                    matched.append(keyword)
        return matched

    async def ai_generate(self, llm_client: LLMClient) -> List[str]:
        """AI 生成关键词"""
        prompt = """
        请生成20个 Telegram 群组引流用的关键词，要求：
        1. 与 VPN/机场/翻墙相关
        2. 包含需求类、咨询类、价格类
        3. 每行一个关键词
        """
        response = await llm_client.generate(prompt)
        return [line.strip() for line in response.split('\n') if line.strip()]
```

class MessageType(Enum):
    GROUP_TEXT = "group_text"
    GROUP_PHOTO = "group_photo"
    PRIVATE_TEXT = "private_text"
    CALLBACK_QUERY = "callback_query"
    COMMAND = "command"


@dataclass
class TelegramMessage:
    """消息对象"""
    message_id: int
    chat_id: int
    sender_id: int
    sender_name: str
    content: str
    message_type: MessageType
    timestamp: datetime
    raw: Dict


class MessageRouter:
    """消息路由器"""
    def __init__(self):
        self.handlers: Dict[MessageType, List[Handler]] = {
            MessageType.GROUP_TEXT: [],
            MessageType.PRIVATE_TEXT: [],
            # ...
        }
        self.rate_limiter = RateLimiter()

    def register(self, message_type: MessageType, handler: Handler):
        """注册处理器"""
        self.handlers[message_type].append(handler)

    async def route(self, message: TelegramMessage):
        """路由消息"""
        # 限流检查
        if not await self.rate_limiter.check(message.chat_id, message.sender_id):
            return

        # 匹配处理器
        handlers = self.handlers.get(message.message_type, [])
        for handler in handlers:
            await handler.handle(message)

    async def process_group_message(self, message: TelegramMessage):
        """处理群组消息"""
        # 关键词匹配
        keywords = await self.keyword_engine.match(message.content)

        if keywords:
            # 触发关键词响应
            for kw in keywords:
                await self.trigger_keyword_response(message, kw)
        else:
            # 正常消息处理
            pass
```

#### 4.5.2 状态定义

```python
# core/user/fsm.py

class UserState(Enum):
    NEW = "new"           # 新注册用户
    PENDING = "pending"   # 待转化（试用期）
    ACTIVE = "active"     # 活跃付费用户
    SILENT = "silent"     # 沉默用户
    CHURNED = "churned"   # 流失用户
    BLOCKED = "blocked"    # 被拉黑用户


class UserFSM:
    """用户状态机"""
    def __init__(self, db: AsyncSession):
        self.db = db
        self.transitions: Dict[Tuple[UserState, str], UserState] = {
            # (当前状态, 事件) -> 新状态
            (UserState.NEW, "trial_started"): UserState.PENDING,
            (UserState.PENDING, "converted"): UserState.ACTIVE,
            (UserState.PENDING, "expired"): UserState.SILENT,
            (UserState.ACTIVE, "expired"): UserState.SILENT,
            (UserState.SILENT, "reactivated"): UserState.ACTIVE,
            (UserState.SILENT, "churned"): UserState.CHURNED,
            (UserState.ANY, "blocked"): UserState.BLOCKED,
        }

    async def transition(self, user: User, event: str):
        """状态转换"""
        current_state = user.state
        new_state = self.transitions.get((current_state, event))

        if new_state:
            old_state = user.state
            user.state = new_state
            await self.db.commit()

            # 触发转换后动作
            await self.on_transition(user, old_state, new_state, event)

    async def on_transition(self, user: User, old: UserState, new: UserState, event: str):
        """转换后动作"""
        if new == UserState.PENDING:
            await self.send_trial_notification(user)

# modules/guardian/moderation/rule_engine.py

class RuleType(Enum):
    KEYWORD = "keyword"
    DOMAIN = "domain"
    FREQUENCY = "frequency"
    IMAGE = "image"


class ViolationLevel(Enum):
    LOW = "low"
    MEDIUM = "medium"
    HIGH = "high"


class ModerationRule:
    """审核规则"""
    def __init__(self, rule_type: RuleType, pattern: str, level: ViolationLevel):
        self.type = rule_type
        self.pattern = pattern
        self.level = level
        self.compiled: Pattern = None

        if rule_type in [RuleType.KEYWORD, RuleType.DOMAIN]:
            self.compiled = re.compile(pattern, re.IGNORECASE)

    def match(self, text: str) -> bool:
        """匹配"""
        if self.type in [RuleType.KEYWORD, RuleType.DOMAIN]:
            return bool(self.compiled.search(text))
        elif self.type == RuleType.FREQUENCY:
            return self.check_frequency(text)
        return False


class RuleEngine:
    """规则引擎"""
    def __init__(self, db: AsyncSession):
        self.db = db
        self.rules: List[ModerationRule] = []
        self.whitelist = WhitelistManager()

    async def load_rules(self):
        """加载规则"""
        # 从数据库加载关键词规则
        keywords = await self.db.query(KeywordRule).all()
        for kw in keywords:
            self.rules.append(ModerationRule(
                RuleType.KEYWORD,
                kw.pattern,
                ViolationLevel(kw.level)
            ))

        # 加载域名规则
        domains = await self.db.query(DomainRule).all()
        for domain in domains:
            self.rules.append(ModerationRule(
                RuleType.DOMAIN,
                domain.pattern,
                ViolationLevel(domain.level)
            ))

    async def evaluate(self, message: TelegramMessage) -> List[ModerationResult]:
        """评估消息"""
        results = []

        # 白名单检查
        if await self.whitelist.is_user_whitelisted(message.sender_id):
            return results

        if await self.whitelist.is_content_whitelisted(message.content):
            return results

        # 规则匹配
        for rule in self.rules:
            if rule.match(message.content):
                results.append(ModerationResult(
                    rule=rule,
                    message=message,
                    level=rule.level
                ))

        return results

    async def execute_actions(self, results: List[ModerationResult]):
        """执行动作"""
        for result in results:
            if result.level == ViolationLevel.HIGH:
                await self.execute_ban(result.message)
            elif result.level == ViolationLevel.MEDIUM:
                await self.execute_mute(result.message)
            else:
                await self.execute_warn(result.message)
```

        self.answer_type = answer_type  # keyword, option, any

class VerificationManager:
    """验证管理器"""
    def __init__(self, db: AsyncSession):
        self.db = db
        self.pending_verifications: Dict[int, VerificationSession] = {}

    async def on_user_join(self, user_id: int, group_id: int):

            return True

        # 创建验证会话
        session = VerificationSession(user_id, group_id, group_config.verification_type)
        self.pending_verifications[user_id] = session

        # 发送验证问题

            return False

            # 验证通过

            if session.attempts >= session.question.max_attempts:
                # 验证失败
                await self.reject_user(user_id, session.group_id)
                del self.pending_verifications[user_id]
                return False

#### 13.1.2 代理池架构

```python
# core/network/proxy_pool.py

class ProxyType(Enum):
    RESIDENTIAL = "residential"   # 住宅代理
    DATACENTER = "datacenter"      # 数据中心
    MOBILE = "mobile"              # 移动代理


@dataclass
class Proxy:
    """代理配置"""
    proxy_id: str
    proxy_type: ProxyType
    host: str
    port: int
    username: str = None          # 认证用户名
    password: str = None          # 认证密码
    protocol: str = "http"        # http/socks5
    is_active: bool = True
    success_rate: float = 1.0      # 成功率
    avg_latency: int = 0          # 平均延迟(ms)
    last_checked: datetime = None


class ProxyPool:
    """代理池管理器"""
    def __init__(self, db: AsyncSession):
        self.db = db
        self.proxies: Dict[str, Proxy] = {}
        self.proxy_groups: Dict[str, List[str]] = {}  # 分组
        self.account_bindings: Dict[str, str] = {}    # 账号-代理绑定
        self._health_check_task = None

    async def add_proxy(self, proxy_config: Proxy) -> str:
        """添加代理"""
        self.proxies[proxy_config.proxy_id] = proxy_config
        await self.db.commit()
        return proxy_config.proxy_id

    async def bind_to_account(self, account_id: str, proxy_id: str):
        """账号绑定代理（固定IP）"""
        self.account_bindings[account_id] = proxy_id

    async def get_proxy(self, account_id: str = None,
                       proxy_type: ProxyType = None) -> Proxy:
        """获取可用代理"""
        # 优先使用绑定代理
        if account_id and account_id in self.account_bindings:
            proxy_id = self.account_bindings[account_id]
            return self.proxies.get(proxy_id)

        # 按类型筛选可用代理
        available = [p for p in self.proxies.values()
                     if p.is_active and p.success_rate > 0.8]

        if proxy_type:
            available = [p for p in available if p.proxy_type == proxy_type]

        # 按成功率和延迟排序
        available.sort(key=lambda x: (-x.success_rate, x.avg_latency))

        return available[0] if available else None

    async def health_check(self):
        """健康检查"""
        for proxy in self.proxies.values():
            try:
                start = time.time()
                async with aiohttp.ClientSession() as session:
                    async with session.get(
                        "https://httpbin.org/ip",
                        proxy=f"{proxy.protocol}://{proxy.host}:{proxy.port}",
                        timeout=10
                    ) as resp:
                        proxy.avg_latency = int((time.time() - start) * 1000)
                        proxy.is_active = resp.status == 200
                        proxy.success_rate = 0.95  # 简化计算
            except:
                proxy.is_active = False
                proxy.success_rate *= 0.9

            proxy.last_checked = datetime.now()

    async def on_proxy_failure(self, proxy_id: str):
        """代理失败处理"""
        proxy = self.proxies.get(proxy_id)
        if proxy:
            proxy.success_rate *= 0.8
            if proxy.success_rate < 0.3:
                proxy.is_active = False
                # 自动切换到备用代理
                await self.switch_to_backup(proxy_id)

    async def switch_to_backup(self, failed_proxy_id: str):
        """切换到备用代理"""
        # 找出使用该代理的所有账号
        affected_accounts = [
            acc for acc, pid in self.account_bindings.items()
            if pid == failed_proxy_id
        ]

        # 为这些账号分配新的可用代理
        for account in affected_accounts:
            new_proxy = await self.get_proxy()
            if new_proxy:
                await self.bind_to_account(account, new_proxy.proxy_id)
```

            user_agent=ua,

违规类型：
- competitor: 竞品推广（VPN/机场/翻墙相关竞品）

- scam: 诈骗内容
- sensitive: 敏感内容
- safe: 安全内容

判断依据：

2. 变体识别：能识别"机+场"、"机勾"等变形词
3. 图片分析：描述图片内容是否违规（如果有）

{"violation": true/false, "type": "违规类型", "confidence": 0.95, "reason": "理由"}

消息内容：
{input}
"""

                     context: List[str] = None) -> ModerationResult:
        """分析消息"""
        prompt = self.moderation_prompt.format(input=message)

        # 添加上下文（前几条消息）
        if context:
            prompt += f"\n\n前几条消息参考：\n{chr(10).join(context[-3:])}"

        response = await self.llm.generate(prompt)
        result = json.loads(response)

        return ModerationResult(
            is_violation=result["violation"],
            violation_type=result["type"],
            confidence=result["confidence"],
            reason=result["reason"],
        )

    async def batch_analyze(self, messages: List[str]) -> List[ModerationResult]:
        """批量分析"""
        tasks = [self.analyze(msg) for msg in messages]

```python
# core/ai/llm_client.py

class LLMProvider(Enum):
    OPENAI = "openai"
    ANTHROPIC = "anthropic"
    LOCAL = "local"  # Ollama


class LLMClient:
    """LLM客户端"""
    def __init__(self, provider: LLMProvider = LLMProvider.OPENAI):
        self.provider = provider
        self.cache = RedisCache()
        self.request_count = 0
        self.total_cost = 0.0

    async def generate(self,
                      prompt: str,
                      model: str = None,
                      temperature: float = 0.7,
                      max_tokens: int = 500) -> str:
        """生成内容"""
        # 检查缓存
        cache_key = hashlib.md5(prompt.encode()).hexdigest()
        cached = await self.cache.get(f"llm:{cache_key}")
        if cached:
            return cached

        # 选择模型
        if not model:
            model = self._get_default_model()

        # 调用API
        if self.provider == LLMProvider.OPENAI:
            response = await self._call_openai(prompt, model, temperature, max_tokens)
        elif self.provider == LLMProvider.ANTHROPIC:
            response = await self._call_anthropic(prompt, model, temperature, max_tokens)
        else:
            response = await self._call_local(prompt, model, temperature, max_tokens)

        # 更新统计
        self.request_count += 1
        self.total_cost += self._calculate_cost(model, max_tokens)

        # 写入缓存（TTL=1小时）
        await self.cache.set(f"llm:{cache_key}", response, ttl=3600)

        return response

    def _get_default_model(self) -> str:
        """获取默认模型"""
        models = {
            LLMProvider.OPENAI: "gpt-4o-mini",
            LLMProvider.ANTHROPIC: "claude-3-haiku-20240307",
            LLMProvider.LOCAL: "qwen2.5:7b",
        }
        return models.get(self.provider, "gpt-4o-mini")

    def _calculate_cost(self, model: str, tokens: int) -> float:
        """计算成本"""
        costs = {
            "gpt-4o": 0.005 * tokens / 1000,
            "gpt-4o-mini": 0.00015 * tokens / 1000,
            "claude-3-sonnet": 0.003 * tokens / 1000,
            "claude-3-haiku": 0.00025 * tokens / 1000,
        }
        return costs.get(model, 0.001)
```
